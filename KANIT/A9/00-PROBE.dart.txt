// DEGIL). G16'nin `sarmalayici`'sini DOGRUDAN kullanir (kendi kurulumunu
// yazmak YASAKTIR, spec kriter 2) -- Dart gorunurlugu dosya-bazli oldugundan
// bu, g16_metin_kaybi_kapisi_test.dart'taki `sarmalayici`'nin PUBLIC
// yapilmasini gerektirdi (o dosyadaki gerekce yorumuna bakiniz).
//
// IKI OLCUM:
//  (1) "Y7: en kucuk N olcumu" -- CakismaCozumSayfasi'nin GOVDE Text'iyle
//      YAPISAL OLARAK AYNI (Scaffold.body -> Center -> Padding(MBosluk.m) ->
//      Text) bir mimik govde, N=1'den baslayarak DOKUZ izgara noktasinin
//      HEPSINDE didExceedMaxLines==false verene kadar artirilir. Mimik
//      kullanilmasinin gerekcesi: `kCakismaGovdesiMaxSatir` DERLEME-ZAMANI
//      sabittir, TEK bir `flutter test` kosumu icinde canli degistirilemez;
//      kriter 4 bu olcumu GERCEK widget uzerinde (G16/_y7) YENIDEN dogrular,
//      mimik yanlissa o adim yakalar.
//  (2) "Y6: rota + semantik olcumu" -- GERCEK CakismaCozumSayfasi'yi
//      Navigator.push ile acar (bu adim PUBLIC gorunurluk GEREKTIRIR) ve
//      320dp'de basligin didExceedMaxLines'ini + semantik etiketinin TAM
//      METNI tasiyip tasimadigini olcer.
//
// Bu iki test SONRASI, kaynagi 00-PROBE.dart.txt olarak arsivlenip dosya
// 0 `test(` kalacak bicimde BOSALTILIR (rm izinli degil, BORCLAR.md:121).
@TestOn('vm')
library;

import 'package:client/design/metinler.dart';
import 'package:client/design/tokens.dart';
import 'package:client/sunum/cakisma_rozeti.dart';
import 'package:flutter/material.dart';
import 'package:flutter/rendering.dart';
import 'package:flutter_test/flutter_test.dart';

import 'g16_metin_kaybi_kapisi_test.dart' show sarmalayici;

const List<double> _olcekler = <double>[1.0, 1.5, 2.0];
const List<double> _genislikler = <double>[320.0, 360.0, 411.0];

/// CakismaCozumSayfasi.build()'in GOVDE Text'iyle YAPISAL OLARAK AYNI mimik
/// -- `sarmalayici` zaten kendi Scaffold'unu sagladigindan burada yalniz
/// Center->Padding->Text govdesi verilir (gercek widget'in body: Center(...)
/// ile BIREBIR ayni yapi).
Widget _y7GovdeMimik(int n) {
  return Builder(
    builder: (context) => Center(
      child: Padding(
        padding: EdgeInsets.all(MBosluk.m),
        child: Text(
          Metinler.cakismaVar,
          style: MTipo.govdeM.copyWith(color: MRenk.metin(context)),
          textAlign: TextAlign.center,
          overflow: TextOverflow.ellipsis,
          maxLines: n,
        ),
      ),
    ),
  );
}

void main() {
  testWidgets('Y7: en kucuk N olcumu (probe)', (tester) async {
    // ignore: avoid_print
    print('KANIT -- GOREV-A9 · Y7 (cakismaVar govdesi) maxLines OLCUMU');
    // ignore: avoid_print
    print('=' * 78);
    // ignore: avoid_print
    print(
      'Yontem: N=1den baslar, DOKUZ izgara noktasinin (320/360/411dp x '
      '1.0/1.5/2.0x) HEPSINDE didExceedMaxLines==false verene kadar artar. '
      'Tavan 8 (N>8 ise DUR); dongu N=12de kosulsuz durur.',
    );
    // ignore: avoid_print
    print('-' * 78);

    int? bulunanN;
    for (var n = 1; n <= 12; n++) {
      var hepsiGecti = true;
      for (final olcek in _olcekler) {
        for (final genislik in _genislikler) {
          await tester.pumpWidget(
            sarmalayici(genislik: genislik, olcek: olcek, bilesen: _y7GovdeMimik(n)),
          );
          await tester.pump();
          final rp = tester.renderObject<RenderParagraph>(find.text(Metinler.cakismaVar));
          final asti = rp.didExceedMaxLines;
          // ignore: avoid_print
          print(
            '  N=$n  ${genislik.toStringAsFixed(0)}dp x ${olcek}x  '
            '-> didExceedMaxLines=$asti  (maxWidth=${rp.constraints.maxWidth})',
          );
          if (asti) hepsiGecti = false;
        }
      }
      if (hepsiGecti) {
        bulunanN = n;
        break;
      }
    }

    // ignore: avoid_print
    print('-' * 78);
    if (bulunanN == null) {
      // ignore: avoid_print
      print('SONUC: [OLCULEMEDI] -- N=12 dahil hicbir deger dokuz noktanin hepsini gecemedi.');
    } else {
      // ignore: avoid_print
      print('SONUC: EN KUCUK N = $bulunanN');
      if (bulunanN > 8) {
        // ignore: avoid_print
        print(
          'DUR: N=$bulunanN tavan(8)\'i asiyor -- Cowork\'e bildirilmeli, '
          'dogru cevap kaydirmadir, bu dilim DEGIL.',
        );
      }
    }
  });

  testWidgets('Y6: rota + semantik olcumu (320dp)', (tester) async {
    tester.view.physicalSize = const Size(320, 640);
    tester.view.devicePixelRatio = 1.0;
    addTearDown(tester.view.reset);

    final tutamac = tester.ensureSemantics();
    await tester.pumpWidget(
      MaterialApp(
        home: Builder(
          builder: (context) => Scaffold(
            body: Center(
              child: ElevatedButton(
                onPressed: () => Navigator.of(context).push(
                  MaterialPageRoute(builder: (_) => const CakismaCozumSayfasi()),
                ),
                child: const Text('ac'),
              ),
            ),
          ),
        ),
      ),
    );
    await tester.tap(find.text('ac'));
    await tester.pumpAndSettle();

    // ignore: avoid_print
    print('KANIT -- GOREV-A9 · Y6 ROTA+SEMANTIK OLCUMU');
    // ignore: avoid_print
    print('=' * 78);
    // ignore: avoid_print
    print(
      'Kurulum: 320dp genislik (tester.view.physicalSize), Navigator.push ile '
      'acilan GERCEK CakismaCozumSayfasi (geri oku VAR, maxLines:1 zaten '
      'gercek widgetta). Ambiyans olcek: 1.0x (test varsayilani, spec bu '
      'olcumde ayri bir izgara istemiyor).',
    );
    // ignore: avoid_print
    print('-' * 78);

    final rp = tester.renderObject<RenderParagraph>(find.text(Metinler.duyuruCakismaVar));
    final didExceed = rp.didExceedMaxLines;
    final semantikVeri = tester.getSemantics(find.text(Metinler.duyuruCakismaVar));
    final tamMi = semantikVeri.label == Metinler.duyuruCakismaVar;

    // ignore: avoid_print
    print('(a) didExceedMaxLines = $didExceed   [HUKUM DEGIL -- OLCUM]');
    // ignore: avoid_print
    print('    maxWidth=${rp.constraints.maxWidth}  size=${rp.size}');
    // ignore: avoid_print
    print('(b) semantik dugum etiketi = "${semantikVeri.label}"');
    // ignore: avoid_print
    print('(b) beklenen TAM metin     = "${Metinler.duyuruCakismaVar}"');
    // ignore: avoid_print
    print('(b) TAM METNI TASIYOR MU   = $tamMi');
    // ignore: avoid_print
    print('-' * 78);
    if (!tamMi) {
      // ignore: avoid_print
      print(
        'DUR: semantik dugum TAM METNI TASIMIYOR -- Y6=1 SAVUNULAMAZ. '
        'Cowork\'e bildirilmeli; dogru cevap sabit degil, basligin '
        'kisaltilmasi ya da bir Semantics tasiyicisidir.',
      );
    } else {
      // ignore: avoid_print
      print(
        'DUR TETIKLENMEDI: semantik dugum tam metni tasiyor -- gorsel kayip '
        '(varsa) yalniz gorseldir, ekran okuyucu etkilenmez.',
      );
    }

    tutamac.dispose();
  });
}
