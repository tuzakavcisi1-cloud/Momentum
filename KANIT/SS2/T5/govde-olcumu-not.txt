/// SS3.1 -- sunucu ile cakisma; dokununca cozum sayfasi acar.
/// Gorunur oldugunda BIR KERELIK semantics duyurusu yapar ("Cakisma var",
/// A11Y-7 / DESIGN.md SS4 -- durum matrisi "cakisma" satiri).
///
/// PAZARLIKSIZ DOKUNMA SINIRI (T6): bu bilesen KENDI GestureDetector'ini
/// tasir ve gorunur metin dugumunun DISINDADIR; GorevSatiri'nin kendisi
/// onTap tasimaz. Gerekce: dokunulabilir alan satir olsaydi metin semantics
/// dugumune girer ve M7 (Semantics etiketi silme mutanti) isirmazdi.
///
/// GOREV-SS2 D-SS2-8: `entityId` + `depo` alir -- cozum sayfasina GECIRIR
/// (M184). `const` kurucu bu yuzden DUSTU (calisma-zamani degeri tasir).
/// `depo` NULLABLE'DIR: `GorevSatiri`'nin BUGUNKU dolayli cagiranlari
/// (`cakismaVarMi: true` ile rozeti yalniz VARLIK icin renderlayan testler --
/// spec SS9/13, yalniz UC satir zorunlu guncelleme gerektirir) depo GECIRMEZ;
/// bu durumda dokunma sessizce hicbir sey yapmaz (sayfa acilmaz), rozet YINE
/// DE dogru RENDERLANIR.
class CakismaRozeti extends StatefulWidget {
  final String entityId;
  final GorevDeposu? depo;

  const CakismaRozeti({super.key, required this.entityId, this.depo});

  @override
  State<CakismaRozeti> createState() => _CakismaRozetiState();
}

class _CakismaRozetiState extends State<CakismaRozeti> {
  bool _duyuruYapildi = false;

  @override
  void didChangeDependencies() {
    super.didChangeDependencies();
    if (_duyuruYapildi) return;
    _duyuruYapildi = true;
    SemanticsService.sendAnnouncement(
      View.of(context),
      Metinler.duyuruCakismaVar,
      TextDirection.ltr,
    );
  }

  @override
  Widget build(BuildContext context) {
    return GestureDetector(
      onTap: () => _cozumSayfasiniAc(context),
      child: Semantics(
        label: Metinler.cakismaVar,
        button: true,
        child: Container(
          width: MOlcu.dokunmaHedefi,
          height: MOlcu.dokunmaHedefi,
          alignment: Alignment.center,
          decoration: BoxDecoration(
            color: MRenk.tehlike(context).withValues(alpha: 0.12),
            borderRadius: BorderRadius.circular(MRadius.s),
          ),
          child: Icon(
            Icons.error_outline,
            size: MOlcu.ikon,
            color: MRenk.tehlike(context),
          ),
        ),
      ),
    );
  }

  void _cozumSayfasiniAc(BuildContext context) {
    final depo = widget.depo;
    if (depo == null) return; // depo verilmemis (dolayli/eski cagiran) -- sessizce hicbir sey yapmaz.
    Navigator.of(context).push(
      MaterialPageRoute(
        builder: (_) => CakismaCozumSayfasi(entityId: widget.entityId, depo: depo),
      ),
    );
  }
}

/// GOREV-SS2 D-SS2-8: `entityId`nin cakisma kayitlarini yan yana gosterir --
/// eski yer tutucu (GOREV-A9 [K93]) burada DOLDURULUR.
///
/// BOS DURUM ZORUNLU: rozet eski `cakisma` kanalindan gelmis olabilir
/// (D-SS2-7) ⇒ 0 kayitla acilis NORMAL haldir; o zaman `Metinler.cakismaKaydiYok`
/// gosterilir ve butonlar GORUNMEZ.
class CakismaCozumSayfasi extends StatefulWidget {
  final String entityId;
  final GorevDeposu depo;

  const CakismaCozumSayfasi({super.key, required this.entityId, required this.depo});

  /// GOREV-A9 [K93/spec SS4/Y6]: 🔒 SABIT, OLCULMEZ. `AppBar` basligi
  /// `_kMaxTitleTextScaleFactor = 1.34`'e kelepceli oldugu icin izgaranin
  /// 1.5x/2.0x ayaklari oraya ULASMAZ; "en kucuk N" kurali uygulansa N=2
  /// cikar ve 2 satir 64 dp toolbar'da SESSIZCE kesilir (spec SS0/2, SS8/S1).
  static const int kCakismaBasligiMaxSatir = 1;

  /// GOREV-SS2 T5: YENIDEN OLCULDU ("korunur" DENMEDI) -- `6` sabiti eski
  /// TAM GENISLIKTE/ORTALANMIS/TEK `Text` duzeni icin OLCULMUSTU (Ö11); bu
  /// dilimde govde YAN YANA/YARIM GENISLIK iki deger bloguna donustu, bu
  /// BASKA bir olcumdur. `16`, `g16_metin_kaybi_kapisi_test.dart` Y7 ayaginin
  /// izgarasinda (olcek 1.0/1.5/2.0 x genislik 320/360/411, yarim genislik)
  /// dokuz noktanin HEPSINDE `didExceedMaxLines==false` veren deger olarak
  /// olculdu (deneme sirasi 3→5→10→16, ilk gecen deger sabitlendi; bkz.
  /// KANIT/SS2/T5-govde-olcumu.txt). Yarim-genislik dar sutunda uzun bir
  /// baslik cok satira yayilabilir -- bu tasarimin BEDELIDIR, kirpma DEGIL.
  static const int kCakismaGovdesiMaxSatir = 16;

  @override
  State<CakismaCozumSayfasi> createState() => _CakismaCozumSayfasiState();
}

class _CakismaCozumSayfasiState extends State<CakismaCozumSayfasi> {
  late Stream<List<CakismaKaydi>> _akis;

  @override
  void initState() {
    super.initState();
    _akis = widget.depo.cakismaKayitlariniIzle(widget.entityId);
  }

  Future<void> _cozVeUygula(CakismaSecimi secim) async {
    await widget.depo.cakismaCoz(widget.entityId, secim);
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text(
          Metinler.duyuruCakismaVar,
          overflow: TextOverflow.ellipsis,
          maxLines: CakismaCozumSayfasi.kCakismaBasligiMaxSatir,
        ),
      ),
      body: StreamBuilder<List<CakismaKaydi>>(
        stream: _akis,
        builder: (context, anlik) {
          final kayitlar = anlik.data ?? const <CakismaKaydi>[];
          if (kayitlar.isEmpty) {
            return Center(
              child: Padding(
                padding: EdgeInsets.all(MBosluk.m),
                child: Text(
                  Metinler.cakismaKaydiYok,
                  textAlign: TextAlign.center,
                  style: MTipo.govdeM.copyWith(color: MRenk.metin(context)),
                  overflow: TextOverflow.ellipsis,
                  maxLines: 3,
                ),
              ),
            );
          }
          return Column(
            children: [
              Expanded(
                child: ListView.builder(
                  padding: EdgeInsets.all(MBosluk.m),
                  itemCount: kayitlar.length,
                  itemBuilder: (context, i) => _kayitSatiri(context, kayitlar[i]),
                ),
              ),
              Padding(
                padding: EdgeInsets.all(MBosluk.m),
                child: Row(
                  children: [
                    Expanded(
                      child: _buton(
                        context,
                        Metinler.cakismaBenimkiniTut,
                        () => _cozVeUygula(CakismaSecimi.benimkiniTut),
                      ),
                    ),
                    SizedBox(width: MBosluk.m),
                    Expanded(
                      child: _buton(
                        context,
                        Metinler.cakismaOnlarinkiniAl,
                        () => _cozVeUygula(CakismaSecimi.onlarinkiniAl),
                      ),
                    ),
                  ],
                ),
              ),
            ],
          );
        },
      ),
    );
  }

  /// A11Y: 48 dp + `Semantics(button: true)` + AYRI etiket (iki buton farkli
  /// dizge tasir -- ekran okuyucu ikisini ayirt edebilmeli).
  Widget _buton(BuildContext context, String etiket, VoidCallback onPressed) {
    return Semantics(
      button: true,
      label: etiket,
      child: SizedBox(
        height: MOlcu.dokunmaHedefi,
        child: ElevatedButton(
          onPressed: onPressed,
          child: Text(etiket, overflow: TextOverflow.ellipsis, maxLines: 1),
        ),
      ),
    );
  }

  Widget _kayitSatiri(BuildContext context, CakismaKaydi kayit) {
    final alanEtiketi = kayit.alan == 'fields:title'
        ? Metinler.cakismaAlaniBaslik
        : Metinler.cakismaAlaniTamamlanma;
    return Padding(
      padding: EdgeInsets.only(bottom: MBosluk.m),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          Text(
            alanEtiketi,
            style: MTipo.etiketS.copyWith(
              fontWeight: FontWeight.w700,
              color: MRenk.metin(context),
            ),
            overflow: TextOverflow.ellipsis,
            maxLines: 1,
          ),
          SizedBox(height: MBosluk.xs),
          Row(
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              Expanded(
                child: _degerBlok(
                  context,
                  Metinler.cakismaBenimki,
                  _degerGoster(kayit.alan, kayit.kaybedenDeger),
                ),
              ),
              SizedBox(width: MBosluk.m),
              Expanded(
                child: _degerBlok(
                  context,
                  Metinler.cakismaOnlarinki,
                  _degerGoster(kayit.alan, kayit.kazananDeger),
                ),
              ),
            ],
          ),
        ],
      ),
    );
  }

  Widget _degerBlok(BuildContext context, String baslik, String deger) {
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Text(
          baslik,
          style: MTipo.etiketS.copyWith(color: MRenk.metinIkincil(context)),
          overflow: TextOverflow.ellipsis,
          maxLines: 1,
        ),
        Text(
          deger,
          style: MTipo.govdeM.copyWith(color: MRenk.metin(context)),
          overflow: TextOverflow.ellipsis,
          maxLines: CakismaCozumSayfasi.kCakismaGovdesiMaxSatir,
        ),
      ],
    );
  }

  /// D-SS2-4: EKRANDA GOSTERILEN METIN kanonik dize DEGILDIR -- yerellestirilmis
  /// etiket kullanilir. `fields:title` icin kanonik dize zaten basligin
  /// KENDISIDIR (D-SS2-4'un "OLDUGU GIBI" kurali).
  String _degerGoster(String alan, String kanonikDeger) {
    if (alan == 'groups:completion') {
      return kanonikDeger == 'tamamlandi'
          ? Metinler.cakismaDegerTamamlandi
          : Metinler.cakismaDegerAcik;
    }
    return kanonikDeger;
  }
}
